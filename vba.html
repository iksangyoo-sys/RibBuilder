Sub dd_Website(control As IRibbonControl, id As String, index As Integer)    Dim url As String     ' 선택한 항목의 id에 따라 이동할 주소 결정    Select Case LCase(Trim(id))        Case "youtube":            url = "https://www.youtube.com"        Case "google":             url = "https://www.google.com"        Case "prompt", "ptompt":   url = "https://iksangyoo-sys.github.io/yes_prom/"        Case Else            MsgBox "알 수 없는 항목입니다: " & id, vbExclamation            Exit Sub    End Select     ' 기본 웹브라우저로 열기    On Error GoTo ErrHandler    ThisWorkbook.FollowHyperlink Address:=url, NewWindow:=True    Exit Sub ErrHandler:    MsgBox "웹사이트를 열 수 없습니다." & vbCrLf & url, vbCriticalEnd Sub

Sub dd_GoFolder(control As IRibbonControl, id As String, index As Integer)    Dim folderPath As String     ' 선택한 항목의 id에 따라 열 폴더 결정    Select Case LCase(Trim(id))        Case "desktop"            ' 바탕화면 (OneDrive로 옮겨진 경우도 자동으로 찾음)            folderPath = CreateObject("WScript.Shell").SpecialFolders("Desktop")        Case "download"            ' 다운로드 폴더            folderPath = Environ("USERPROFILE") & "\Downloads"            If Dir(folderPath, vbDirectory) = "" Then folderPath = "shell:Downloads"        Case Else            MsgBox "알 수 없는 항목입니다: " & id, vbExclamation            Exit Sub    End Select     ' 파일 탐색기로 폴더 열기    On Error GoTo ErrHandler    Shell "explorer.exe """ & folderPath & """", vbNormalFocus    Exit Sub ErrHandler:    MsgBox "폴더를 열 수 없습니다." & vbCrLf & folderPath, vbCriticalEnd Sub

비밀번호 : 애플리케이션 비밀번호

나머지 중요하지 않은 항목은 일반적인 설정(기본값)으로 처리해도 괜찮습니다.

Option Explicit '========= 여기만 수정하세요! =============Const SMTP_USER As String = "yessoee"                    ' 네이버 아이디Const SMTP_PASS As String = "여기에_애플리케이션_비밀번호"   ' 네이버 2단계 인증 애플리케이션 비밀번호Const FROM_ADDR As String = "yessoee@naver.com"          ' 보내는 사람 주소 (로그인 계정과 같아야 함)Const FROM_NAME As String = "유익상"                      ' 보내는 사람 표시 이름Const START_ROW As Long = 2                              ' 데이터 시작 행 (1행은 제목줄)Const WAIT_SEC As Long = 2                               ' 발송 간격(초)'====================================== Sub 네이버메일발송()    Dim ws As Worksheet: Set ws = ActiveSheet    Dim lastRow As Long, r As Long    Dim toAddr As String, subj As String, body As String    Dim total As Long, okCnt As Long, failCnt As Long, skipCnt As Long     lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row    If lastRow < START_ROW Then        MsgBox "보낼 데이터가 없습니다.", vbExclamation        Exit Sub    End If     ' 보낼 건수 세기 (이메일이 있고, 아직 성공하지 않은 행)    For r = START_ROW To lastRow        If Trim(ws.Cells(r, "A").Value) <> "" And Left(ws.Cells(r, "D").Value, 2) <> "성공" Then total = total + 1    Next r    If total = 0 Then        MsgBox "새로 보낼 메일이 없습니다. (모두 발송 완료)", vbInformation        Exit Sub    End If    If MsgBox("총 " & total & "건을 발송합니다. 진행할까요?", vbYesNo + vbQuestion) <> vbYes Then Exit Sub     If ws.Range("D1").Value = "" Then ws.Range("D1").Value = "발송결과"    Application.Cursor = xlWait     For r = START_ROW To lastRow        toAddr = Trim(ws.Cells(r, "A").Value)        subj = CStr(ws.Cells(r, "B").Value)        body = Replace(CStr(ws.Cells(r, "C").Value), vbLf, vbCrLf)   ' 셀 줄바꿈 유지         If toAddr = "" Then            ' 빈 행은 건너뜀        ElseIf Left(ws.Cells(r, "D").Value, 2) = "성공" Then            skipCnt = skipCnt + 1        ElseIf Not toAddr Like "?*@?*.?*" Then            ws.Cells(r, "D").Value = "실패 " & Format(Now, "yyyy-mm-dd hh:nn:ss") & " (이메일 형식 오류)"            failCnt = failCnt + 1        Else            Application.StatusBar = "메일 발송 중... " & (okCnt + failCnt + 1) & " / " & total & "  " & toAddr            Dim errMsg As String            errMsg = SendNaverMail(toAddr, subj, body)            If errMsg = "" Then                ws.Cells(r, "D").Value = "성공 " & Format(Now, "yyyy-mm-dd hh:nn:ss")                okCnt = okCnt + 1            Else                ws.Cells(r, "D").Value = "실패 " & Format(Now, "yyyy-mm-dd hh:nn:ss") & " (" & errMsg & ")"                failCnt = failCnt + 1            End If            DoEvents            Application.Wait Now + TimeSerial(0, 0, WAIT_SEC)   ' 스팸 차단 방지용 간격        End If    Next r     Application.StatusBar = False    Application.Cursor = xlDefault    MsgBox "발송 완료" & vbCrLf & _           "성공: " & okCnt & "건" & vbCrLf & _           "실패: " & failCnt & "건" & vbCrLf & _           "건너뜀(이미 성공): " & skipCnt & "건", vbInformationEnd Sub ' 메일 1건 발송. 성공하면 "" , 실패하면 오류 내용을 돌려줌Private Function SendNaverMail(toAddr As String, subj As String, body As String) As String    Const CFG As String = "http://schemas.microsoft.com/cdo/configuration/"    Dim msg As Object, conf As Object     On Error GoTo Fail    Set conf = CreateObject("CDO.Configuration")    With conf.Fields        .Item(CFG & "sendusing") = 2                    ' 외부 SMTP 서버 사용        .Item(CFG & "smtpserver") = "smtp.naver.com"        .Item(CFG & "smtpserverport") = 465        .Item(CFG & "smtpusessl") = True                ' SSL 보안 연결        .Item(CFG & "smtpauthenticate") = 1             ' 아이디/비밀번호 인증        .Item(CFG & "sendusername") = SMTP_USER        .Item(CFG & "sendpassword") = SMTP_PASS        .Item(CFG & "smtpconnectiontimeout") = 30        .Update    End With     Set msg = CreateObject("CDO.Message")    With msg        Set .Configuration = conf        .BodyPart.Charset = "utf-8"                     ' 한글 깨짐 방지        .From = """" & FROM_NAME & """ <" & FROM_ADDR & ">"        .To = toAddr        .Subject = subj        .TextBody = body        .Send    End With    SendNaverMail = ""    Exit Function Fail:    SendNaverMail = Err.DescriptionEnd Function
